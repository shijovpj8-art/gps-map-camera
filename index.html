// CAPTURE PHOTOGRAPH WITH QR CODE function capturePhoto() { if (!stream || video.readyState < 2) { alert("Please start the camera first."); return; } if (latitude === null || longitude === null) { alert("Please get GPS location before capturing the photograph."); return; } if (typeof QRCode === "undefined") { alert("QR code library not loaded. Check your internet connection."); return; } const width = video.videoWidth; const height = video.videoHeight; if (width === 0 || height === 0) { alert("Camera is not ready. Please try again."); return; } canvas.width = width; canvas.height = height; const ctx = canvas.getContext("2d"); // Draw original photograph ctx.drawImage(video, 0, 0, width, height); const now = new Date(); const mapsURL = "https://www.google.com/maps?q=" + latitude + "," + longitude; // Generate QR code const qrContainer = document.getElementById("qrcode"); qrContainer.innerHTML = ""; const qrSize = Math.round( Math.min(width * 0.12, height * 0.10) ); new QRCode(qrContainer, { text: mapsURL, width: qrSize, height: qrSize, correctLevel: QRCode.CorrectLevel.H }); const qrCanvas = qrContainer.querySelector("canvas"); if (!qrCanvas) { alert("Unable to generate QR code. Please try again."); return; } // Compact information strip const padding = width * 0.025; const fontSize = Math.max(14, width * 0.017); const labelSize = Math.max(10, fontSize * 0.65); const qrLabelHeight = labelSize * 1.5; const stripHeight = Math.ceil( Math.max( qrSize + qrLabelHeight + padding * 2, fontSize * 3.5 + padding * 2 ) ); const stripTop = height - stripHeight; ctx.fillStyle = "rgba(0,0,0,0.78)"; ctx.fillRect(0, stripTop, width, stripHeight); // GPS coordinates ctx.fillStyle = "#ffffff"; ctx.font = "bold " + fontSize + "px Arial"; ctx.textAlign = "left"; ctx.fillText( "GPS: " + latitude.toFixed(6) + ", " + longitude.toFixed(6), padding, stripTop + padding + fontSize ); // Date and time ctx.font = fontSize + "px Arial"; ctx.fillText( "Date: " + now.toLocaleDateString() + " | Time: " + now.toLocaleTimeString(), padding, stripTop + padding + fontSize * 2.5 ); // QR code at bottom-right const qrX = width - qrSize - padding; const qrY = stripTop + (stripHeight - qrSize - qrLabelHeight) / 2; ctx.fillStyle = "#ffffff"; ctx.fillRect( qrX - 4, qrY - 4, qrSize + 8, qrSize + 8 ); ctx.drawImage( qrCanvas, qrX, qrY, qrSize, qrSize ); // QR label ctx.fillStyle = "#8ff0ad"; ctx.font = "bold " + labelSize + "px Arial"; ctx.textAlign = "center"; ctx.fillText( "SCAN MAP", qrX + qrSize / 2, qrY + qrSize + labelSize * 1.2 ); ctx.textAlign = "left"; // Prepare photograph for saving photo = canvas.toDataURL("image/jpeg", 0.95); document.getElementById("photoStatus").textContent = "Photograph captured successfully. Ready to save."; }
